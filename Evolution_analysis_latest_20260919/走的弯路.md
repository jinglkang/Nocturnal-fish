```slim_bla.pl
#!/usr/bin/perl
use strict;
use warnings;

my $bla=$ARGV[0];
my %hash;
open BLA, $bla or die "can not open $bla\n";
while (<BLA>) {
        chomp;
        my @a=split /\t/;
        my ($spe, $id, $len);
        $id=$a[0]; $len=$a[4];
        ($spe)=$a[0]=~/(.*?)\_/;
        if ($hash{$spe}) {
                my $oldlen=$hash{$spe}->{'LEN'};
                my $newlen=$len;
                if ($newlen > $oldlen) {
                        $hash{$spe}={
                        'ID'  => $id,
                        'LEN' => $len,
                        'INF' => $_
                        };
                }
        } else {
                $hash{$spe}={
                        'ID'  => $id,
                        'LEN' => $len,
                        'INF' => $_
                        };
        }
}

foreach my $spe (sort keys %hash) {
        my $info=$hash{$spe}->{'INF'};
        print "$info\n";
}
```
